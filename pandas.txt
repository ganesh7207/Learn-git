import seaborn as sns
import matplotlib.pyplot as plt

# Load the built-in Titanic dataset
titanic = sns.load_dataset("titanic")

# Display first 5 records
print(titanic.head(5))

# Remove rows where age or fare is missing
data = titanic.dropna(subset=["age", "fare", "pclass"])

# Create scatter plot
plt.scatter(
    data["age"],
    data["fare"],
    c=data["pclass"],
    cmap="viridis"
)

# Add title and labels
plt.title("Titanic Passenger Age vs Fare")
plt.xlabel("Age")
plt.ylabel("Fare")

# Add color bar to show passenger class
plt.colorbar(label="Passenger Class")

# Display the plot
plt.show()

import pandas as pd
import matplotlib.pyplot as plt

# -------------------------------------------------
# Step 1: Create the dictionary
# -------------------------------------------------

profit_data = {
    "Development": [200, 195, 250],
    "ILP": [400, 418, 431],
    "CT": [195, 150, 188],
    "ITIS": [200, 230, 270],
    "Others": [150, 155, 17]
}

# -------------------------------------------------
# Step 2: Create DataFrame
# Years are used as the index
# -------------------------------------------------

df = pd.DataFrame(
    profit_data,
    index=[2017, 2018, 2019]
)

print("Profit Data:")
print(df)

# -------------------------------------------------
# Step 3: Create grouped bar chart
# -------------------------------------------------

ax = df.plot(
    kind="bar",
    figsize=(10, 6)
)

# -------------------------------------------------
# Step 4: Customize the chart
# -------------------------------------------------

plt.title("Department-wise Profit Percentage (2017-2019)")
plt.xlabel("Year")
plt.ylabel("Profit Percentage")

plt.legend(
    title="Department",
    loc="upper left"
)

# Background color
plt.gca().set_facecolor("xkcd:silver")

# Add grid
plt.grid(
    axis="y",
    linestyle="--",
    alpha=0.5
)

# Adjust layout
plt.tight_layout()

# -------------------------------------------------
# Step 5: Display the chart
# -------------------------------------------------

plt.show()

import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# -------------------------------------------------
# Step 1: Load the dataset
# -------------------------------------------------

file_path = "marks.csv"

df = pd.read_csv(file_path, sep=',')

# Display first few records
print(df.head())

# -------------------------------------------------
# Step 2: Create the boxplot
# -------------------------------------------------

sns.boxplot(
    x="sub1",
    y="sub2",
    data=df
)

# -------------------------------------------------
# Step 3: Add title and labels
# -------------------------------------------------

plt.title("Age Distribution by Education Level")
plt.xlabel("Education Level")
plt.ylabel("Age")

# -------------------------------------------------
# Step 4: Display the plot
# -------------------------------------------------

plt.show()

import matplotlib.pyplot as plt
import pandas as pd
import seaborn as sns

df = pd.read_csv("student_scores.csv")

# Plotting the three subject score distributions side-by-side
plt.figure(figsize=(8, 5))
sns.boxplot(data=df[["Math", "Science", "English"]])
plt.title("Subject Score Distribution")
plt.ylabel("Scores")
plt.xlabel("Subjects")
plt.grid(axis="y", linestyle="--", alpha=0.7)
plt.show()
